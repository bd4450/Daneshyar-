// ۱. دیتابیس محلی (برای سوالات پرتکرار و آفلاین) const localKnowledge = { "تهران": "تهران پایتخت ایران و بزرگترین شهر این کشور است.", "دماوند": "بلندترین قله ایران و بلندترین آتشفشان خاورمیانه.", "آب": "مایعی حیاتی با فرمول شیمیایی H2O که در دمای ۱۰۰ درجه می‌جوشد." // سوالات خود را اینجا به همین شکل اضافه کنید }; // ۲. تابع اصلی جستجو (هسته برنامه) async function getAnswer(query) { // پاکسازی ورودی (حذف فاصله‌های اضافی) const q = query.trim(); // مرحله اول: چک کردن دیتابیس محلی if (localKnowledge[q]) { return { source: "محلی", text: localKnowledge[q] }; } // مرحله دوم: جستجو در ویکی‌پدیا (در صورت آنلاین بودن) try { const response = await fetch(`https://fa.wikipedia.org/api/rest_v1/page/summary/${encodeURIComponent(q)}`, { // تنظیمات برای پایداری در اینترنت ضعیف signal: AbortSignal.timeout(5000) }); if (!response.ok) throw new Error("Not found"); const data = await response.json(); if (data.extract) { return { source: "ویکی‌پدیا", text: data.extract }; } } catch (error) { console.log("خطا در اتصال به ویکی‌پدیا یا کلمه یافت نشد."); } // مرحله سوم: در صورتی که هیچکدام پیدا نشد return { source: "هیچ‌کدام", text: "متأسفانه پاسخی برای این سوال یافت نشد." }; } // ۳. نحوه استفاده در دکمه جستجو async function handleSearch() { const input = document.getElementById("searchBox").value; const resultDisplay = document.getElementById("resultDisplay"); resultDisplay.innerText = "در حال جستجو..."; // نمایش وضعیت بارگذاری const result = await getAnswer(input); // نمایش نتیجه در صفحه resultDisplay.innerHTML = `

${result.text}

منبع: ${result.source}
`; }
