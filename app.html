function renderResults(results) { let html = ""; // نمایش معانی چندگانه لغت‌نامه‌ها if (results.exactMatch) { html += `

معانی واژه: ${results.exactMatch.word}

`; results.exactMatch.sources.forEach(s => { html += `

${s.dict}: ${s.meaning}

`; }); html += `
`; } // نمایش پاسخ‌های ویژه جدول و اطلاعات عمومی if (results.crosswordMatches.length > 0) { html += `

راهنمای حل جدول و اصطلاحات:

    `; results.crosswordMatches.forEach(c => { html += `
  • ${c.clue}: ${c.answer}
    • `; }); html += `
`; } // نمایش کلمات مرتبط (در صورتی که دقیقاً معنا مرتبط باشد) if (results.relatedMeanings.length > 0) { html += `

کلمات مرتبط در متن لغت‌نامه‌ها:

    `; results.relatedMeanings.slice(0, 5).forEach(r => { html += `
  • ${r.word} (${r.dict}): ${r.snippet}
    • `; }); html += `
`; } if (!results.exactMatch && results.crosswordMatches.length === 0 && results.relatedMeanings.length === 0) { html = `
موردی یافت نشد. لطفاً املا را بررسی کنید.
`; } document.getElementById("resultsContainer").innerHTML = html; }
